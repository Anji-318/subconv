# 1. 装 bun（一次性，全局安装）
npm install -g bun

# 2. 验证
bun --version

# 3. 装项目依赖
bun install

# 4. 构建产物
bun run pages:build

# 5. 部署到 Cloudflare Pages
npx wrangler pages deploy .vercel\output\static --project-name=subconv
